npm ci
npm run build
# 在后台环境配置 TYPESAFE_API_KEY
# Xmax: XMAX_API_KEY，XMAX_REGION=china 或 global
# Decart: DECART_API_KEY（标准模式）
# 阿里云: DASHSCOPE_API_KEY，DASHSCOPE_BASE_URL（选填）
npm start
